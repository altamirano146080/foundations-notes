\section{Apuntes del Tema 5: Multilayer Perceptron (MLP I y II)}

\subsection{1. De Modelos Lineales a No Lineales}

\subsubsection{1.1 Limitaciones de la Transformación Afín}

\begin{itemize}
	\item \textbf{Modelos Lineales:} Modelos como la Regresión Lineal o la Regresión Logística calculan transformaciones afines de la forma $z = x^\top w + b$.
	
	\item \textbf{El Problema del Colapso Lineal:} Si se apilan múltiples capas compuestas únicamente por transformaciones afines, la red colapsa matemáticamente en un único modelo lineal equivalente.
	
	\item \emph{Demostración:} Dadas dos capas lineales $H = X W^{(1)} + b^{(1)}$ y $O = H W^{(2)} + b^{(2)}$:
	$$O = (X W^{(1)} + b^{(1)}) W^{(2)} + b^{(2)} = X (W^{(1)} W^{(2)}) + (b^{(1)} W^{(2)} + b^{(2)}) = X W + b$$
	donde $W = W^{(1)} W^{(2)}$ y $b = b^{(1)} W^{(2)} + b^{(2)}$. La capacidad expresiva no aumenta con respecto a un modelo de una sola capa.
	
	\item \textbf{Separabilidad No Lineal (Problema XOR):} Las fronteras de decisión rectilíneas son incapaces de separar clases complejas o disjuntas (como la función lógica XOR o patrones circulares/concéntricos).
\end{itemize}

\subsection{2. Capas Ocultas y Funciones de Activación}

Para dotar al Perceptrón Multicapa (MLP) de la capacidad de aprender relaciones no lineales, se aplica una \textbf{función de activación no lineal} $\phi(\cdot)$ elemento a elemento tras cada transformación afín en las capas ocultas.

\begin{lstlisting}[language=Python]
	Entrada X (n x d) ---> [ Z = X.W(1) + b(1) ] ---> [ H = phi(Z) ] ---> Salida O
\end{lstlisting}

\subsubsection{2.1 Principales Funciones de Activación}

\begin{table}[htbp]
	\centering
	\small
	\begin{tabular}{|p{2.5cm}|p{3.2cm}|p{2.2cm}|p{5cm}|}
		\hline
		\textbf{Función} & \textbf{Ecuación $\phi(z)$} & \textbf{Rango} & \textbf{Propiedades y Derivada $\phi'(z)$} \\ \hline
		\textbf{Sigmoide} & $\sigma(z) = \frac{1}{1 + e^{-z}}$ & $(0, 1)$ & Interpretación probabilística. \textbf{Problema:} Zonas llanas donde $\sigma'(z) \approx 0$, provocando desvanecimiento de gradiente (\emph{vanishing gradient}). \\ \hline
		\textbf{Tanh} & $\tanh(z) = \frac{e^z - e^{-z}}{e^z + e^{-z}}$ & $(-1, 1)$ & Centrada en cero. Sufre de desvanecimiento para $\vert z\vert$ elevados ($\tanh'(z) = 1 - \tanh^2(z)$). \\ \hline
		\textbf{ReLU} & $\text{ReLU}(z) = \max(0, z)$ & $[0, \infty)$ & \textbf{Estándar en redes profundas:} Eficiente. Derivada $\phi'(z) = 1$ para $z > 0$. \textbf{Riesgo:} \emph{Dying ReLU} si $z \le 0$ permanentemente. \\ \hline
	\end{tabular}
\end{table}

\subsection{3. Clasificación Multiclase: Regresión Softmax}

Cuando el target $y$ pertenece a una de las $K$ categorías posibles ($K > 2$) sin un orden explícito, se emplea la codificación \textbf{One-Hot Encoding} ($y \in \mathbb{R}^K$).

\subsubsection{3.1 Operador Softmax}

La capa de salida genera $K$ valores no acotados denominados \textbf{logits} ($O = [o_1, o_2, \dots, o_K]$). Para transformar los logits en una distribución de probabilidad bien definida donde $\sum_{k=1}^{K} \hat{y}_k = 1$ y $\hat{y}_k > 0$, se aplica la función \textbf{Softmax}:
$$\hat{y}_k = \text{Softmax}(o_k) = \frac{e^{o_k}}{\sum_{j=1}^{K} e^{o_j}}$$

\begin{itemize}
	\item \textbf{Propiedad Conservadora del Orden:} Preserva la ordenación monótona de los logits ($\arg\max_k \hat{y}_k = \arg\max_k o_k$).
\end{itemize}

\subsubsection{3.2 Pérdida de Entropía Cruzada Categorical (\emph{Categorical Cross-Entropy})}

Para una muestra con vector One-Hot $y$ y predicciones $\hat{y}$:
$$\mathcal{L}(y, \hat{y}) = - \sum_{k=1}^{K} y_k \log(\hat{y}_k)$$

\begin{quote}
	\textbf{Nota de Implementación (PyTorch):} En la práctica, no se añade manualmente la capa \texttt{Softmax} al final del modelo. Se retorna el tensor de logits directamente y se utiliza \texttt{nn.CrossEntropyLoss()}, la cual combina internamente \texttt{Softmax} y \texttt{Cross-Entropy} de forma numéricamente estable mediante el truco \emph{log-sum-exp}.
\end{quote}

\subsection{4. Algoritmo de Propagación: Forward Pass y Backpropagation}

\subsubsection{4.1 Forward Pass (Propagación hacia Adelante)}

Para un MLP de $L$ capas con entrada $X \in \mathbb{R}^{n \times d}$ ($H^{(0)} = X$):
Para $i = 1, 2, \dots, L-1$:
$$Z^{(i)} = H^{(i-1)} W^{(i)} + b^{(i)}$$
$$H^{(i)} = \phi(Z^{(i)})$$
Salida final ($L$):
$$O = H^{(L-1)} W^{(L)} + b^{(L)}$$

\subsubsection{4.2 Backpropagation (Propagación hacia Atrás)}

Consiste en la aplicación sistemática de la \textbf{Regla de la Cadena} multivariante desde la función de pérdida $\mathcal{J}$ hasta las primeras capas para obtener los gradientes $\frac{\partial \mathcal{J}}{\partial W^{(i)}}$ y $\frac{\partial \mathcal{J}}{\partial b^{(i)}}$:
$$\frac{\partial \mathcal{J}}{\partial W^{(i)}} = (H^{(i-1)})^\top \cdot \delta^{(i)}, \quad \text{donde } \delta^{(i)} = \frac{\partial \mathcal{J}}{\partial Z^{(i)}}$$

\begin{itemize}
	\item \textbf{Gestión de Memoria:} Los tensores intermedios calculados durante el \emph{forward pass} ($H^{(i)}, Z^{(i)}$) deben almacenarse en memoria para ser reevaluados durante el \emph{backpropagation}, por lo que entrenar consume considerablemente más memoria que ejecutar inferencia (\texttt{with torch.no\_grad():}).
\end{itemize}

\subsection{5. Propiedades Teóricas e Inicialización de Parámetros}

\subsubsection{5.1 Teorema de Aproximación Universal}

\begin{itemize}
	\item \textbf{Definición:} Un Perceptrón Multicapa con tan solo \textbf{una capa oculta} y un número suficiente de neuronas no lineales puede aproximar cualquier función continua en un dominio compacto con precisión arbitraria $\epsilon > 0$.
	
	\item \textbf{¿Por qué usar redes profundas?} El teorema demuestra la \emph{existencia} teórica, pero no especifica la cantidad de neuronas requeridas (que puede ser masiva/exponencial) ni si se pueden aprender eficazmente. Añadir profundidad permite representar funciones complejas mediante composiciones jerárquicas con un \textbf{número drásticamente menor de parámetros}.
\end{itemize}

\subsubsection{5.2 Rompimiento de la Simetría e Inicialización}

\begin{itemize}
	\item \textbf{Inconveniente de Inicialización en Cero:} Si todos los pesos de una capa oculta se inicializan con la misma constante (p. ej. $w_{ij} = 0$), todas las neuronas calcularán el mismo valor en el \emph{forward} y recibirán el mismo gradiente en el \emph{backward}. El MLP colapsaría al comportamiento de un modelo lineal único.
	
	\item \textbf{Estrategias de Inicialización:}
	\begin{itemize}
		\item \textbf{Xavier / Glorot Initialization:} Diseñada para activaciones sigmoideas o $\tanh$. Mantiene constante la varianza a través de las capas:
		$$W \sim \mathcal{N}\left(0, \frac{2}{n_{\text{in}} + n_{\text{out}}}\right)$$
		\item \textbf{He / Kaiming Initialization:} Diseñada para activaciones \textbf{ReLU}, compensando que la mitad de las neuronas se desactivan ($\le 0$):
		$$W \sim \mathcal{N}\left(0, \frac{2}{n_{\text{in}}}\right)$$
	\end{itemize}
\end{itemize}

\subsection{6. Problemas en Redes Profundas: Desvanecimiento y Explosión del Gradiente}

Al encadenar múltiples capas mediante la regla de la cadena, la magnitud del gradiente acumulado en las capas iniciales se calcula como un producto iterativo:
$$\frac{\partial \mathcal{J}}{\partial W^{(1)}} \propto \prod_{l=2}^{L} W^{(l)} \cdot \phi'(Z^{(l)})$$

\begin{itemize}
	\item \textbf{Vanishing Gradients (Desvanecimiento):} Si las derivadas o valores propios son menores a $1$ (p. ej., sigmoide/tanh en saturación), el producto decrece exponencialmente hacia $0$. Las capas iniciales dejan de aprender.
	\item \textbf{Exploding Gradients (Explosión):} Si los valores de los pesos son mayores a $1$, el producto crece exponencialmente provocando inestabilidad numérica (\texttt{NaN} o \texttt{Inf}). Se mitiga aplicando \textbf{Gradient Clipping} (\texttt{nn.utils.clip\_grad\_norm\_}).
\end{itemize}

\subsection{7. Práctica y Aplicación en Código (PyTorch)}

\begin{lstlisting}[language=Python]
	import torch
	import torch.nn as nn
	import torch.optim as optim
	
	class DeepMLP(nn.Module):
	def __init__(self, input_dim=8, hidden_dims=[12, 6], num_classes=3, dropout_rate=0.2):
	super(DeepMLP, self).__init__()
	
	layers = []
	in_dim = input_dim
	
	for h_dim in hidden_dims:
	linear_layer = nn.Linear(in_dim, h_dim)
	nn.init.kaiming_normal_(linear_layer.weight, nonlinearity='relu')
	nn.init.zeros_(linear_layer.bias)
	
	layers.append(linear_layer)
	layers.append(nn.ReLU())
	layers.append(nn.Dropout(p=dropout_rate))
	in_dim = h_dim
	
	out_layer = nn.Linear(in_dim, num_classes)
	nn.init.xavier_normal_(out_layer.weight)
	nn.init.zeros_(out_layer.bias)
	layers.append(out_layer)
	
	self.network = nn.Sequential(*layers)
	
	def forward(self, x):
	return self.network(x)
	
	def train_mlp_multiclass():
	device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
	
	X_dummy = torch.randn(100, 8)
	y_dummy = torch.randint(0, 3, (100,))
	
	model = DeepMLP(input_dim=8, hidden_dims=[12, 6], num_classes=3).to(device)
	
	criterion = nn.CrossEntropyLoss()
	optimizer = optim.Adam(model.parameters(), lr=0.005)
	
	X_b = X_dummy.to(device)
	y_b = y_dummy.to(device)
	
	model.train()
	for epoch in range(1, 101):
	optimizer.zero_grad()
	logits = model(X_b)
	loss = criterion(logits, y_b)
	loss.backward()
	
	nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
	optimizer.step()
	
	if epoch % 25 == 0:
	print(f"Epoca [{epoch:3d}/100] | Loss: {loss.item():.4f}")
	
	if __name__ == "__main__":
	train_mlp_multiclass()
\end{lstlisting}

\subsection*{Pregunta} ¿Qué representa un logit en clasificación multiclase?
\textbf{Solución.} La puntuación sin normalizar de una clase antes de aplicar la transformación usada por la pérdida o la probabilidad.

\subsection*{Pregunta} ¿Qué calcula autograd?
\textbf{Solución.} Construye y recorre el grafo de operaciones para obtener derivadas respecto de los tensores que requieren gradiente.
